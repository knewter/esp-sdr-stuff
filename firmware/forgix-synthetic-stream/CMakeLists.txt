cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "")
set(PICO_BOARD pico2 CACHE STRING "MCU profile only, not physical FPGA qualification")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "")
set(PICO_SDK_FETCH_FROM_GIT OFF CACHE BOOL "")
set(PICO_TINYUSB_PATH "$ENV{PICO_SDK_PATH}/lib/tinyusb" CACHE PATH "")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(forgix_synthetic_stream C CXX ASM)
if(NOT DEFINED ENV{PIOASM_EXECUTABLE})
 message(FATAL_ERROR "Pinned Nix pioasm required")
endif()
add_executable(pioasm IMPORTED GLOBAL)
set_property(TARGET pioasm PROPERTY IMPORTED_LOCATION "$ENV{PIOASM_EXECUTABLE}")
set(TINYUSB_OPT_OS OPT_OS_NONE)
pico_sdk_init()
add_executable(forgix_synthetic_stream main.c platform.c wire_packet.c engine.c codec.c protocol.c config.c uid.c usb_descriptors.c fpga_image.c)
target_include_directories(forgix_synthetic_stream PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_BINARY_DIR})
target_compile_definitions(forgix_synthetic_stream PRIVATE BRIDGE_EMBEDDED_CONFIG=1
 PICO_STACK_SIZE=4096 PICO_HEAP_SIZE=0 PICO_STDIO_UART=0 PICO_STDIO_USB=0 PICO_USE_GPIO_COPROCESSOR=0)
target_compile_options(forgix_synthetic_stream PRIVATE -Wall -Wextra -Werror)
target_link_options(forgix_synthetic_stream PRIVATE "LINKER:--undefined=diagnostic_profile" "LINKER:--undefined=stream_layout")
target_link_libraries(forgix_synthetic_stream pico_stdlib pico_bootrom tinyusb_device hardware_watchdog hardware_clocks hardware_pio)
pico_enable_stdio_uart(forgix_synthetic_stream 0)
pico_enable_stdio_usb(forgix_synthetic_stream 0)
pico_set_binary_type(forgix_synthetic_stream no_flash)
pico_add_map_output(forgix_synthetic_stream)
pico_generate_pio_header(forgix_synthetic_stream ${CMAKE_CURRENT_LIST_DIR}/wire.pio)
