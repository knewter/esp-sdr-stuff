cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "")
set(PICO_BOARD pico2 CACHE STRING "SDK MCU clock profile only; not a Forgix pin claim")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "")
set(PICO_SDK_FETCH_FROM_GIT OFF CACHE BOOL "")
set(PICO_TINYUSB_PATH "$ENV{PICO_SDK_PATH}/lib/tinyusb" CACHE PATH "")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(forgix_usb_ram C CXX ASM)
set(TINYUSB_OPT_OS OPT_OS_NONE)
pico_sdk_init()
add_executable(forgix_usb_ram main.c usb_descriptors.c)
target_include_directories(forgix_usb_ram PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_BINARY_DIR})
target_compile_definitions(forgix_usb_ram PRIVATE PICO_STACK_SIZE=4096 PICO_HEAP_SIZE=0
  PICO_STDIO_UART=0 PICO_STDIO_USB=0 PICO_USE_GPIO_COPROCESSOR=0)
target_compile_options(forgix_usb_ram PRIVATE -Wall -Wextra -Werror)
target_link_libraries(forgix_usb_ram pico_stdlib tinyusb_device hardware_watchdog)
pico_enable_stdio_uart(forgix_usb_ram 0)
pico_enable_stdio_usb(forgix_usb_ram 0)
pico_set_binary_type(forgix_usb_ram no_flash)
pico_add_map_output(forgix_usb_ram)
# Export only the ELF/map. No UF2, BIN, flash command or programming task.
