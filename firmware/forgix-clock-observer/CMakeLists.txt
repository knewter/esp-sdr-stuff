cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "")
set(PICO_BOARD pico2 CACHE STRING "SDK MCU clock profile only")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "")
set(PICO_SDK_FETCH_FROM_GIT OFF CACHE BOOL "")
set(PICO_TINYUSB_PATH "$ENV{PICO_SDK_PATH}/lib/tinyusb" CACHE PATH "")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(forgix_clock_observer C CXX ASM)
if(NOT DEFINED ENV{PIOASM_EXECUTABLE} OR NOT FORGIX_CONFIG_IMAGE)
 message(FATAL_ERROR "Pinned assembler and exact observer image required")
endif()
add_executable(pioasm IMPORTED GLOBAL)
set_property(TARGET pioasm PROPERTY IMPORTED_LOCATION "$ENV{PIOASM_EXECUTABLE}")
set(TINYUSB_OPT_OS OPT_OS_NONE)
pico_sdk_init()
add_executable(forgix_clock_observer main.c engine.c observer.c rp_input.c config.c
 protocol.c usb_descriptors.c uid.c ${FORGIX_CONFIG_IMAGE})
target_include_directories(forgix_clock_observer PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_BINARY_DIR})
target_compile_definitions(forgix_clock_observer PRIVATE PICO_STACK_SIZE=4096 PICO_HEAP_SIZE=0
 PICO_STDIO_UART=0 PICO_STDIO_USB=0 PICO_USE_GPIO_COPROCESSOR=0)
target_compile_options(forgix_clock_observer PRIVATE -Wall -Wextra -Werror)
target_link_options(forgix_clock_observer PRIVATE "LINKER:--undefined=diagnostic_profile" "LINKER:--undefined=clock_layout")
target_link_libraries(forgix_clock_observer pico_stdlib pico_bootrom tinyusb_device
 hardware_watchdog hardware_clocks hardware_pio)
pico_enable_stdio_uart(forgix_clock_observer 0)
pico_enable_stdio_usb(forgix_clock_observer 0)
pico_set_binary_type(forgix_clock_observer no_flash)
pico_add_map_output(forgix_clock_observer)
pico_generate_pio_header(forgix_clock_observer ${CMAKE_CURRENT_LIST_DIR}/period.pio)
# ELF/map only. No flash, UF2, BIN or load task.
